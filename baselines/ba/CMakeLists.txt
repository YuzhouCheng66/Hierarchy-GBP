cmake_minimum_required(VERSION 3.21)
project(measured_ba_wrappers LANGUAGES CXX)

# This project builds only the archived entry points. RootBA and all third-party
# libraries are supplied by the user, never downloaded or copied into the release.
set(ROOTBA_SOURCE "" CACHE PATH "Pinned, patched RootBA source tree")
set(ROOTBA_BUILD "" CACHE PATH "Matching Release static-library RootBA build")
set(ROOTBA_LIB_CONFIG "Release" CACHE STRING "Library subdirectory for multi-config builds")
if(NOT EXISTS "${ROOTBA_SOURCE}/src/rootba/bal/solver_options.hpp" OR NOT IS_DIRECTORY "${ROOTBA_BUILD}")
  message(FATAL_ERROR "Set ROOTBA_SOURCE and ROOTBA_BUILD; see README.md")
endif()
if(CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_CONFIGURATION_TYPES Release CACHE STRING "Release-only imported libraries" FORCE)
elseif(NOT CMAKE_BUILD_TYPE OR NOT CMAKE_BUILD_TYPE STREQUAL "Release")
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Release-only imported libraries" FORCE)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)

find_package(Eigen3 3.4.0 EXACT REQUIRED CONFIG)
find_package(Ceres 2.2.0 EXACT REQUIRED CONFIG)
find_package(glog REQUIRED CONFIG)
find_package(fmt REQUIRED CONFIG)
find_package(absl REQUIRED CONFIG)
find_package(TBB REQUIRED CONFIG)

add_library(rootba_headers INTERFACE)
target_include_directories(rootba_headers INTERFACE "${ROOTBA_SOURCE}/src")
foreach(path Sophus visit_struct/include wise_enum enum-flags/include nameof/include
             toml11 json basalt-headers/include magic_enum pprint cereal/include)
  target_include_directories(rootba_headers INTERFACE "${ROOTBA_SOURCE}/external/${path}")
endforeach()
target_compile_definitions(rootba_headers INTERFACE NDEBUG ROOTBA_DISABLE_ASSERTS
  ROOTBA_INSTANTIATIONS_DOUBLE ROOTBA_INSTANTIATIONS_STATIC_LMB ROOTBA_BUILD_QR CEREAL_THREAD_SAFE=1)
target_link_libraries(rootba_headers INTERFACE Eigen3::Eigen glog::glog fmt::fmt
  absl::flat_hash_map TBB::tbb)
if(MSVC)
  target_compile_options(rootba_headers INTERFACE /O2 /W3 /Zc:__cplusplus /bigobj /utf-8)
  # oneTBB headers can auto-link tbb12.lib in addition to TBB::tbb's tbb.lib.
  get_target_property(tbb_location TBB::tbb IMPORTED_IMPLIB_RELEASE)
  if(tbb_location)
    get_filename_component(tbb_lib_dir "${tbb_location}" DIRECTORY)
    target_link_directories(rootba_headers INTERFACE "${tbb_lib_dir}")
  endif()
endif()

function(import_rootba component subdir)
  find_library(${component}_LIBRARY NAMES ${component}
    PATHS "${ROOTBA_BUILD}/src/rootba/${subdir}/${ROOTBA_LIB_CONFIG}"
          "${ROOTBA_BUILD}/src/rootba/${subdir}"
    NO_DEFAULT_PATH REQUIRED)
  add_library(${component} STATIC IMPORTED)
  set_target_properties(${component} PROPERTIES IMPORTED_LOCATION "${${component}_LIBRARY}")
endfunction()
import_rootba(rootba "")
import_rootba(rootba_cli cli)
import_rootba(rootba_ceres ceres)
import_rootba(rootba_solver solver)
import_rootba(rootba_qr qr)
import_rootba(rootba_sc sc)
import_rootba(rootba_cg cg)
target_link_libraries(rootba INTERFACE rootba_headers)
target_link_libraries(rootba_cli INTERFACE rootba)
target_link_libraries(rootba_ceres INTERFACE rootba Ceres::ceres)
target_link_libraries(rootba_qr INTERFACE rootba)
target_link_libraries(rootba_sc INTERFACE rootba)
target_link_libraries(rootba_cg INTERFACE rootba)
target_link_libraries(rootba_solver INTERFACE rootba_cg rootba_sc rootba_qr rootba)

  add_executable(bal_ceres_${kind}_schur src/bal_ceres.cpp)

  target_compile_definitions(bal_ceres_${kind}_schur PRIVATE ROOTBA_FORCE_CERES_${kind_upper}_SCHUR)
