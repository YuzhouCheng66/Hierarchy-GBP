cmake_minimum_required(VERSION 3.20)
project(MeasuredPGOBaselineWrappers LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
find_package(OpenMP REQUIRED)

set(PGO_G2O_ROOT "" CACHE PATH "External g2o install for CHOLMOD/Schwarz, with include/lib/bin")
set(PGO_EIGEN_ROOT "" CACHE PATH "Matching Eigen include root for CHOLMOD/Schwarz")
set(PGO_G2O_OPENMP_ROOT "" CACHE PATH "Separate external OpenMP-enabled g2o install for PCG")
set(PGO_PCG_EIGEN_ROOT "" CACHE PATH "Eigen include root matching the PCG g2o ABI")
option(PGO_ARCHIVED_AVX2 "Use the archived CHOLMOD/Schwarz MSVC AVX2 code-generation flag" ON)

function(pgo_wrapper target source group root eigen)
    if(NOT EXISTS "${root}/include/g2o/config.h" OR NOT EXISTS "${eigen}/Eigen/Core")
        message(FATAL_ERROR "${target}: explicit matching g2o and Eigen roots are required")
    endif()
    add_executable(${target} "src/${source}")
    target_include_directories(${target} PRIVATE "src" "${eigen}" "${root}/include")
    target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(${target} PRIVATE OpenMP::OpenMP_CXX)
    foreach(lib g2o_core g2o_stuff ${ARGN})
        find_library(${target}_${lib} NAMES ${lib} PATHS "${root}/lib" NO_DEFAULT_PATH REQUIRED)
        target_link_libraries(${target} PRIVATE "${${target}_${lib}}")
    endforeach()
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/${group}")
    foreach(config RELEASE DEBUG RELWITHDEBINFO MINSIZEREL)
        set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/${group}")
    endforeach()
    if(MSVC)
        target_compile_options(${target} PRIVATE /O2 /Ob2 /DNDEBUG /bigobj)
        target_link_options(${target} PRIVATE /OPT:REF /OPT:ICF /INCREMENTAL:NO)
        if(PGO_ARCHIVED_AVX2 AND NOT group STREQUAL "pcg16")
            target_compile_options(${target} PRIVATE /arch:AVX2)
        endif()
    else()
        target_compile_options(${target} PRIVATE -O3 -DNDEBUG)
    endif()
    if(WIN32)
        # Never mix the different archived g2o/CHOLMOD/OpenMP runtime bundles.
        file(GLOB runtime_dlls "${root}/bin/*.dll")
        foreach(dll IN LISTS runtime_dlls)
            add_custom_command(TARGET ${target} POST_BUILD
                COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${dll}" "$<TARGET_FILE_DIR:${target}>"
                VERBATIM)
        endforeach()
    endif()
endfunction()

if(PGO_G2O_ROOT)
    foreach(space se2 se3)
        pgo_wrapper(cholmod_${space} cholmod_frontend.cpp cholmod1 "${PGO_G2O_ROOT}" "${PGO_EIGEN_ROOT}" g2o_solver_cholmod)
        if(space STREQUAL "se2")
            target_compile_definitions(cholmod_${space} PRIVATE DIRECT_SE2)
        endif()
        pgo_wrapper(g2o_block_pcg_${space} ${space}_main.cpp schwarz16 "${PGO_G2O_ROOT}" "${PGO_EIGEN_ROOT}" g2o_solver_pcg)
    endforeach()
endif()
if(PGO_G2O_OPENMP_ROOT)
    pgo_wrapper(g2o_pcg_openmp pcg_frontend.cpp pcg16 "${PGO_G2O_OPENMP_ROOT}" "${PGO_PCG_EIGEN_ROOT}" g2o_solver_pcg)
    target_compile_definitions(g2o_pcg_openmp PRIVATE EIGEN_DONT_PARALLELIZE)
    if(MSVC)
        target_compile_options(g2o_pcg_openmp PRIVATE /Ox /Oi /wd4251 /wd4244 /wd4267)
    endif()
endif()
if(NOT PGO_G2O_ROOT AND NOT PGO_G2O_OPENMP_ROOT)
    message(FATAL_ERROR "Supply PGO_G2O_ROOT and/or PGO_G2O_OPENMP_ROOT; dependencies are never fetched implicitly")
endif()
