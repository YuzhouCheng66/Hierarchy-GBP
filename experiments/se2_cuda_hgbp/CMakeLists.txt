cmake_minimum_required(VERSION 3.24)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
set(CMAKE_CUDA_ARCHITECTURES 89 CACHE STRING "CUDA architectures")
project(SE2CudaHGBP LANGUAGES CXX CUDA)
if(NOT WIN32)
    message(FATAL_ERROR "The validated host backend currently requires Windows/MSVC.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD 17)
set(HGBP_BUILD_BA OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
find_package(OpenMP REQUIRED COMPONENTS CXX)
get_filename_component(REFERENCE "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_subdirectory("${REFERENCE}" reference)

# Keep the released CPU16 target intact. The duplicate host library supplies
# its exact residual implementation under a different entry name for shadow checks.
get_target_property(REFERENCE_SOURCES hgbp_se2 SOURCES)
set(HOST_SOURCES)
foreach(source IN LISTS REFERENCE_SOURCES)
    if(source STREQUAL "src/se2_internal_residual.cpp")
        list(APPEND HOST_SOURCES residual_reference.cpp)
    else()
        list(APPEND HOST_SOURCES "${REFERENCE}/${source}")
    endif()
endforeach()
add_library(se2_cuda_host STATIC ${HOST_SOURCES})
target_include_directories(se2_cuda_host PUBLIC "${REFERENCE}/include" ${HGBP_EIGEN_DIR}
    ${HGBP_VCPKG_INSTALLED_DIR}/include ${HGBP_VCPKG_INSTALLED_DIR}/include/suitesparse)
target_link_directories(se2_cuda_host PUBLIC ${HGBP_VCPKG_INSTALLED_DIR}/lib)
target_link_libraries(se2_cuda_host PUBLIC OpenMP::OpenMP_CXX PRIVATE
    cholmod suitesparseconfig amd colamd camd ccolamd lapack openblas)
hgbp_compile(se2_cuda_host)
target_compile_options(se2_cuda_host PRIVATE /Zi /Zo /Oy-)
set_source_files_properties(residual_reference.cpp PROPERTIES COMPILE_OPTIONS /fp:fast)

add_library(se2_cuda_kernels STATIC se2_cuda.cu)
target_compile_options(se2_cuda_kernels PRIVATE
    $<$<COMPILE_LANGUAGE:CUDA>:-O3;--fmad=true;--prec-div=true;--prec-sqrt=true;-lineinfo;-Xptxas=-v>)
add_library(se2_cuda_solver OBJECT benchmark.cpp se2_checked_bridge.cpp se2_overlap_solver.cpp
    se2_host_overlap.cpp se2_basis_workspace.cpp se2_cached_coarse.cpp)
target_link_libraries(se2_cuda_solver PRIVATE se2_cuda_host)
hgbp_compile(se2_cuda_solver)
add_executable(se2_cuda_benchmark $<TARGET_OBJECTS:se2_cuda_solver>)
target_link_libraries(se2_cuda_benchmark PRIVATE se2_cuda_host se2_cuda_kernels)
hgbp_compile(se2_cuda_benchmark)

enable_testing()
add_executable(se2_cuda_failure_test test_se2_segment_failure.cpp se2_host_overlap.cpp)
target_link_libraries(se2_cuda_failure_test PRIVATE se2_cuda_kernels)
add_executable(se2_cuda_layout_test test_se2_transfer_layout.cpp)
add_executable(se2_cuda_overlap_test test_se2_host_overlap.cpp se2_host_overlap.cpp)
add_executable(se2_cuda_coarse_test test_se2_coarse_cache.cpp se2_cached_coarse.cpp)
target_link_libraries(se2_cuda_coarse_test PRIVATE se2_cuda_host)
foreach(target se2_cuda_failure_test se2_cuda_layout_test se2_cuda_overlap_test se2_cuda_coarse_test)
    hgbp_compile(${target})
    add_test(NAME ${target} COMMAND ${target})
endforeach()
foreach(target se2_cuda_benchmark se2_cuda_failure_test se2_cuda_layout_test se2_cuda_overlap_test se2_cuda_coarse_test)
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/pgo")
endforeach()
